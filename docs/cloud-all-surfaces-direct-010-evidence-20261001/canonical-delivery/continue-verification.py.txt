"""Continue the existing shipment verification; never rebuild or overwrite initial evidence."""
from pathlib import Path

DRIVER = Path(__file__).resolve().parent / 'run-verification.py'
# Reuse only reviewed constants and function definitions preceding driver startup.
PREFIX = DRIVER.read_text().split('\nEVIDENCE.mkdir(parents=True, exist_ok=True)')[0]
exec(compile(PREFIX, str(DRIVER), 'exec'), globals())

assert DELIVERY.exists()
assert not PRIVATE.exists()
PRIVATE.mkdir()
for folder in ['config', 'tmp', 'profiles', 'cache/bun', 'cache/npm', 'cache/xdg', 'cache/node']:
    (PRIVATE / folder).mkdir(parents=True, exist_ok=True)
ENV = {'PATH': str(pathlib.Path(BUN).parent) + ':' + os.environ.get('PATH', ''),
       'LANG': 'C.UTF-8', 'LC_ALL': 'C.UTF-8', 'TZ': 'UTC', 'TMPDIR': str(PRIVATE / 'tmp'),
       'ROX_CONFIG_DIR': str(PRIVATE / 'config'), 'CRAFT_CONFIG_DIR': str(PRIVATE / 'config'),
       'BUN_INSTALL_CACHE_DIR': str(PRIVATE / 'cache/bun'), 'npm_config_cache': str(PRIVATE / 'cache/npm'),
       'XDG_CACHE_HOME': str(PRIVATE / 'cache/xdg'), 'NODE_COMPILE_CACHE': str(PRIVATE / 'cache/node'),
       'ELECTRON_SKIP_BINARY_DOWNLOAD': '1', 'NO_COLOR': '1', 'FORCE_COLOR': '0'}
for name in ['HTTP_PROXY', 'HTTPS_PROXY', 'NO_PROXY', 'http_proxy', 'https_proxy', 'no_proxy',
             'SSL_CERT_FILE', 'REQUESTS_CA_BUNDLE', 'NODE_EXTRA_CA_CERTS', 'NODE_OPTIONS']:
    if name in os.environ:
        ENV[name] = os.environ[name]
RESULTS = []
before = json.loads((EVIDENCE / 'source-before.json').read_text())
original_before = json.loads((EVIDENCE / 'original-source-before.json').read_text())
live_before = json.loads((EVIDENCE / 'live-artifacts-before.json').read_text())
shipment_before = json.loads((EVIDENCE / 'shipment-before-runtime-manifest.json').read_text())
initial_receipt = json.loads((EVIDENCE / 'verification-receipt.json').read_text())
assert source(CHECKOUT) == before and source(ORIGINAL) == original_before
assert artifacts(['apps/webui/dist', 'dist-server']) == live_before
save('continuation-provenance.json', {'candidate': EXPECTED, 'initialReceiptSha256': digest(EVIDENCE / 'verification-receipt.json'),
    'initialShipmentManifestSha256': digest(EVIDENCE / 'shipment-before-runtime-manifest.json'),
    'originalDriverSha256': digest(DRIVER), 'sourceAndOriginalEqualInitialBefore': True,
    'buildRerun': False, 'uiStagingRerun': False, 'initialEvidenceOverwritten': False,
    'initialHarnessFailure': {'command': initial_receipt['results'][-1]['argv'], 'exit': 2,
        'reason': 'observer file was not yet written', 'applicationFailure': False,
        'driverLine': 172, 'retainedReceipt': 'shipment-closure-receipt.json', 'retainedLog': 'shipment-closure.log'},
    'outerBun': subprocess.check_output([BUN, '--version'], env=ENV).decode().strip(),
    'node': subprocess.check_output(['node', '--version'], env=ENV).decode().strip(),
    'safeEnvironmentKeys': sorted(ENV), 'HOMERepurposed': False})

error = None
shipment_after = None
try:
    run('canonical-delivery-smoke', [BUN, '--no-env-file', 'test', str(EVIDENCE / 'canonical-delivery-smoke.test.ts')])
    run('shipment-closure-retry', ['python3', str(EVIDENCE / 'verify-shipment-closure.py')])
    run('manifest-after-runtime', ['python3', str(EVIDENCE / 'canonical-delivery-manifest.py'), 'after', EXPECTED])
    shipment_after = json.loads((EVIDENCE / 'canonical-delivery-after.json').read_text())
    assert shipment_before['files'] == shipment_after['files']
    assert shipment_before['symlinks'] == shipment_after['symlinks']
    phases = json.loads((EVIDENCE / 'canonical-delivery-phases.json').read_text())
    assert phases['compilerTestBun'] == '1.3.14' and phases['observedShippedBun'] == '1.3.9'
    assert phases['profilesRemoved'] and len(phases['servers']) == 3
    assert [s['exit'] for s in phases['servers']] == [0, 0, 1]
    assert all(not s['tokenWasLogged'] and not s['sigkillFallback'] for s in phases['servers'])
    assert all(p['pass'] for p in phases['phases']) and phases['assertions'] == 56
    assert RESULTS[0]['actualBunCounts'] == {'pass': 2, 'fail': 0, 'expect': 56, 'tests': 2, 'files': 1}
    helper = (EVIDENCE / 'canonical-delivery-smoke.test.ts').read_text()
    smoke_lines = (EVIDENCE / 'canonical-delivery-smoke.log').read_text().splitlines()
    case_rows = []
    for index, match in enumerate(re.finditer(r"\bit\('([^']+)'", helper)):
        name = match.group(1)
        declaration_line = helper[:match.start()].count('\n') + 1
        actual = [(n + 1, line) for n, line in enumerate(smoke_lines) if '(pass)' in line and name in line]
        assert len(actual) == 1
        case_rows.append({'name': name, 'testFile': 'canonical-delivery-smoke.test.ts',
            'declarationLine': declaration_line, 'actualRawLogLine': actual[0][0],
            'actualRawLogText': actual[0][1], 'result': 'pass',
            'phaseAssertionCount': 47 if index == 0 else 9})
    assert len(case_rows) == 2
    save('executed-cases.json', {'candidate': EXPECTED, 'actualCases': case_rows,
        'actualBunCounts': RESULTS[0]['actualBunCounts'], 'helperAssertionCount': phases['assertions'],
        'testNamesExtractedFromDeclarationsAndMatchedToActualPassLog': True,
        'logSha256': digest(EVIDENCE / 'canonical-delivery-smoke.log')})
except BaseException as exc:
    error = str(exc)
    raise
finally:
    after, original_after = source(CHECKOUT), source(ORIGINAL)
    live_after = artifacts(['apps/webui/dist', 'dist-server'])
    save('source-after-continuation.json', after)
    save('original-source-after-continuation.json', original_after)
    save('live-artifacts-after-continuation.json', live_after)
    phases_path = EVIDENCE / 'canonical-delivery-phases.json'
    phases = json.loads(phases_path.read_text()) if phases_path.exists() else None
    cleanup = {'sourceAndLockPreserved': before == after, 'originalSourceAndStatusPreserved': original_before == original_after,
        'originalCleanBeforeAndAfter': original_before['status'] == original_after['status'] == '',
        'liveArtifactsPreserved': live_before == live_after, 'liveArtifactFileCount': live_after['fileCount'],
        'ownChildProcessesExited': bool(phases) and all(s['exit'] is not None for s in phases['servers']),
        'ownProfilesRemovedByUnchangedHelper': bool(phases) and phases['profilesRemoved'],
        'canonicalOutputRetainedForRootVerification': DELIVERY.exists(),
        'sharedCheckoutDependenciesAndArtifactsRetained': True,
        'sourceRegularFileCount': before['regularFileCount'], 'gitlinkCount': before['gitlinkCount'],
        'privateTemporaryRootRemoved': False,
        'shipmentRegularFilesAndSymlinksUnchangedDuringRuntime': bool(shipment_after) and
            shipment_before['files'] == shipment_after['files'] and shipment_before['symlinks'] == shipment_after['symlinks']}
    if not phases or cleanup['ownChildProcessesExited']:
        shutil.rmtree(PRIVATE)
        cleanup['privateTemporaryRootRemoved'] = not PRIVATE.exists()
    save('cleanup-receipt-final.json', cleanup)
    save('final-verification-receipt.json', {'candidate': EXPECTED, 'results': RESULTS, 'error': error,
        'initialHarnessFailureRetainedAndResolved': True, 'applicationFailureObserved': False if error is None else None,
        'canonicalBuildReceipt': initial_receipt['results'][1], 'initialReceiptUnchanged': True,
        'cleanup': cleanup, 'runtimePhases': phases,
        'scopeLimitations': ['no installed/native Electron or macOS', 'no provider or OAuth', 'no iOS',
                            'no PG', 'no full109/full143', 'no security scan acceptance']})
    print(json.dumps({'error': error, 'cleanup': cleanup}, ensure_ascii=False), flush=True)
