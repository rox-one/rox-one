def digest(path):
    h = hashlib.sha256()
    with pathlib.Path(path).open('rb') as stream:
        for part in iter(lambda: stream.read(1024 * 1024), b''):
            h.update(part)
    return h.hexdigest()

def digest_bytes(data):
    return hashlib.sha256(data).hexdigest()

def save(name, value):
    (EVIDENCE / name).write_text(json.dumps(value, indent=2, ensure_ascii=False) + '\n')

def git(repo, *args):
    return subprocess.check_output(['git', '-C', str(repo), *args]).decode().strip()

def source(repo):
    files, links, gitlinks = {}, {}, {}
    for record in filter(None, subprocess.check_output(['git', '-C', str(repo), 'ls-files', '-s', '-z']).decode().split('\0')):
        meta, name = record.split('\t', 1)
        mode, object_id, stage = meta.split()
        if mode == '160000':
            gitlinks[name] = object_id
        elif mode == '120000':
            target = os.readlink(repo / name)
            links[name] = {'target': target, 'sha256': digest_bytes(target.encode())}
        else:
            files[name] = digest(repo / name)
    return {'head': git(repo, 'rev-parse', 'HEAD'), 'status': git(repo, 'status', '--porcelain'),
            'regularFileCount': len(files), 'gitlinkCount': len(gitlinks), 'symlinkCount': len(links),
            'files': files, 'gitlinks': gitlinks, 'symlinks': links}

def artifacts(folders):
    files = {}
    for folder in folders:
        for file in sorted((CHECKOUT / folder).rglob('*')):
            if file.is_file():
                files[str(file.relative_to(CHECKOUT))] = {'bytes': file.stat().st_size, 'sha256': digest(file)}
    return {'fileCount': len(files), 'files': files,
            'manifestSha256': digest_bytes(json.dumps(files, sort_keys=True, separators=(',', ':')).encode())}

def run(label, argv):
    log = EVIDENCE / (label + '.log')
    start = time.monotonic()
    began = datetime.datetime.now(datetime.timezone.utc).isoformat()
    with log.open('wb') as stream:
        completed = subprocess.run(argv, cwd=CHECKOUT, env=ENV, stdout=stream, stderr=subprocess.STDOUT)
    receipt = {'label': label, 'argv': argv, 'cwd': str(CHECKOUT), 'startedUtc': began,
               'seconds': round(time.monotonic() - start, 3), 'exit': completed.returncode,
               'log': str(log), 'logSha256': digest(log), 'safeEnvironmentKeys': sorted(ENV)}
    output = log.read_text(errors='replace')
    counts = re.search(r'\s+(\d+) pass\s+\n\s*(\d+) fail\s+\n\s*(\d+) expect\(\) calls\s+\nRan (\d+) tests across (\d+) files', output)
    if counts:
        receipt['actualBunCounts'] = dict(zip(['pass', 'fail', 'expect', 'tests', 'files'], map(int, counts.groups())))
    RESULTS.append(receipt)
    save(label + '-receipt.json', receipt)
    print(json.dumps(receipt, ensure_ascii=False), flush=True)
    if completed.returncode:
        raise RuntimeError(label + ' failed; inspect retained safe log')
    return receipt
