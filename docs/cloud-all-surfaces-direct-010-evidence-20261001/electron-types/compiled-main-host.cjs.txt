function denied(category) {
  stats[category]++;
  const error = new Error(`Controlled host probe refused ${category}`);
  error.code = 'CLOUD_PROBE_GUARD';
  throw error;
}
function inside(base, target) { const relative = path.relative(base, target); return relative === '' || (!relative.startsWith('..') && !path.isAbsolute(relative)); }
function guardFile(value, write) {
  if (typeof value === 'number') return;
  const target = path.resolve(value instanceof URL ? fileURLToPath(value) : String(value));
  if (!write && target === '/proc/version') { stats.publicSystemMetadataReads++; return; }
  if (!(write ? inside(profile, target) : inside(task, target))) {
    stats.outsideFileDetails.push({ operation: write ? 'write' : 'read',
      target: target.startsWith('/etc/') ? target : `<outside-task>/${path.basename(target)}`,
      credentialLikePath: /credential|auth|token|secret|oauth|config\.ya?ml|\.env/i.test(target),
      caller: new Error('outside file guard').stack });
    denied(write ? 'outsideWrites' : 'outsideReads');
  }
}
for (const name of ['readFileSync', 'readFile']) {
  const original = fs[name]; fs[name] = function(file, ...args) { guardFile(file, false); return original.call(this, file, ...args); };
}
for (const name of ['writeFileSync', 'writeFile', 'appendFileSync', 'appendFile', 'mkdirSync', 'mkdir', 'rmSync', 'rm', 'unlinkSync', 'unlink']) {

    guardFile(file, write); return original.call(this, file, flags, ...args);
  };
}

let finished = false;
function finish(error) {
  if (finished) return; finished = true;
  const result = { bundleSha256: bundleHash, actualNode: process.versions.node, mockedElectronHost: true,
    declaredMockElectronVersion: '39.2.7', installedElectronOrMacAcceptance: false, readinessHeld: true,
    ...stats, error: error ? { code: error.code, message: error.message, stack: error.stack } : null };
  originalStdout(JSON.stringify(result) + '\n');
  process.exit(error ? 1 : 0);
}
