DIST=CHECKOUT/'apps/electron/dist'
def sha(data):return hashlib.sha256(data).hexdigest()
tracked=set(subprocess.check_output(['git','ls-files','-z'],cwd=CHECKOUT).decode().split('\0'))
textual=[];wrappers=[];inventory=[];map_artifacts=[]
required=json.loads(pathlib.Path('/workspace/rox-r15-evidence-20260930/union-validation/a4ca-electron-summary.json').read_text())['required_shell_sources_compiled']
required={k:[] for k in required}
maps=sorted(DIST.glob('renderer/**/*.map'))
for p in maps:
 d=json.loads(p.read_text());sources=d['sources'];contents=d.get('sourcesContent',[])
 assert len(sources)==len(contents),p
 map_artifacts.append({'path':str(p.relative_to(CHECKOUT)),'bytes':p.stat().st_size,'sha256':sha(p.read_bytes())})

   entry['generatedExportDefaultWrapper']=content is not None and bool(re.fullmatch(r'export default "(?:data:image/[^\n]+|__VITE_ASSET__[A-Za-z0-9_$-]+__)";?',content))
   wrappers.append(entry)
  else:textual.append(entry)
  suffix=relative.removeprefix('apps/electron/src/renderer/')
  if suffix in required:required[suffix].append(entry['chunk'])
  if '/pages/' in relative or '/platform/' in relative:inventory.append({'chunk':entry['chunk'],'source':relative})
artifacts=[]
for p in sorted(DIST.rglob('*')):
