function redact(value) {
  let s = String(value)
  for (const token of secrets) s = s.replaceAll(token, '[ephemeral test token redacted]')
  return s.replace(/CRAFT_SERVER_TOKEN=[^\r\n]*/g, 'CRAFT_SERVER_TOKEN=[redacted]')
    .replace(/((?:Cookie|Set-Cookie|Authorization)\s*[:=])[^\r\n]*/gi, '$1[redacted]')
    .replace(/(\b(?:accessToken|refreshToken|password|token)\s*[=:]\s*["']?)[^\s,"'}]+/gi, '$1[redacted]')
}
function safeValues(value) {
  if (typeof value === 'string') return redact(value)
  if (Array.isArray(value)) return value.map(safeValues)
  if (value && typeof value === 'object') return Object.fromEntries(Object.entries(value).map(([k, v]) => [k, safeValues(v)]))
  return value
}

async function within(promise, ms, message) {
  let timer
  try { return await Promise.race([promise, new Promise((_, reject) => { timer = setTimeout(() => reject(new Error(message)), ms) })]) }
  finally { clearTimeout(timer) }
}

async function reservePort() {
  const server = net.createServer(); await new Promise((resolve, reject) => { server.once('error', reject); server.listen(0, '127.0.0.1', resolve) })
  const port = server.address().port; await new Promise(resolve => server.close(resolve)); return port
}
async function spawnServer(profile, label, sameProfile = {}) {
  const token = sameProfile.token || crypto.randomBytes(32).toString('hex'); secrets.add(token)
  const port = sameProfile.port || await reservePort()
  const child = cp.spawn(bun, [path.join(checkout, 'dist-server/index.js')], {
    cwd: checkout, detached: true,
    env: { PATH: path.dirname(bun) + ':' + (process.env.PATH || ''), LANG: 'C.UTF-8', TMPDIR: path.join(runRoot, 'tmp'),
      ROX_CONFIG_DIR: profile, CRAFT_CONFIG_DIR: profile, ROX_SERVER_TOKEN: token, CRAFT_SERVER_TOKEN: token,
      CRAFT_RPC_HOST: '127.0.0.1', CRAFT_RPC_PORT: String(port), CRAFT_HEALTH_PORT: '0',
      CRAFT_WEBUI_DIR: path.join(checkout, 'apps/webui/dist'), CRAFT_BROWSER_BACKEND: 'none',
      CRAFT_BUNDLED_ASSETS_ROOT: path.join(checkout, 'apps/electron'), CRAFT_DISABLE_MESSAGING: 'true' },
    stdio: ['ignore', 'pipe', 'pipe'],
  })
  trackChild(child)
  let output = ''; let readyResolve; const ready = new Promise(resolve => { readyResolve = resolve })
  const exited = new Promise(resolve => { child.once('exit', (code, signal) => resolve({ code, signal })); child.once('error', error => resolve({ error: error.message })) })
  const closed = childClosePromises.get(child)
  function collect(chunk) { output += chunk.toString(); if (output.includes(`CRAFT_SERVER_URL=ws://127.0.0.1:${port}`)) readyResolve() }
  child.stdout.on('data', collect); child.stderr.on('data', collect)
  const audit = { label, pid: child.pid, profile, origin: `http://127.0.0.1:${port}`, sentSigterm: false, usedSigkill: false }
  result.servers.push(audit)
  const server = { child, exited, origin: audit.origin, token, audit, profile,
    async stop() {
      if (audit.exit) return
      audit.preStopExitCode = child.exitCode
      if (child.exitCode === null && child.signalCode === null) { audit.sentSigterm = child.kill('SIGTERM') }
      try { audit.exit = await within(exited, 10_000, 'UI baseline server SIGTERM timeout') }
      catch (error) { audit.usedSigkill = true; try { process.kill(-child.pid, 'SIGKILL') } catch { child.kill('SIGKILL') } audit.exit = await within(exited, 5000, 'Owned server did not exit after SIGKILL'); audit.stopError = String(error) }
      await within(closed, 5000, 'Owned server did not close after stop'); fs.writeFileSync(path.join(runRoot, `${label}-server.log`), redact(output)); audit.fullEphemeralTokenLogged = output.includes(token)
      children.delete(child); checkpoint()
    },
  }
  try { await within(Promise.race([ready, exited.then(e => { throw new Error(`Server exited before readiness: ${JSON.stringify(e)}`) })]), 45_000, 'UI server readiness timeout') }
  catch (error) { await server.stop(); throw new Error(`${error}\n${redact(output)}`) }
  return server
}

function observe(page, label) {
  const log = { label, pageErrors: [], consoleErrors: [], consoleWarnings: [], failedRequests: [], badResponses: [], webSockets: [] }
  result.pages.push(log)
  page.on('pageerror', error => log.pageErrors.push(redact(error.stack || error.message)))
  page.on('console', msg => { if (msg.type() === 'error') log.consoleErrors.push(redact(msg.text())); if (msg.type() === 'warning') log.consoleWarnings.push(redact(msg.text())) })
  page.on('requestfailed', req => log.failedRequests.push({ url: redact(req.url()), method: req.method(), error: req.failure()?.errorText }))
  page.on('response', response => { if (response.status() >= 400) log.badResponses.push({ url: redact(response.url()), status: response.status() }) })
  page.on('websocket', ws => {
    const requests = new Map()
    const socket = { url: redact(ws.url()), sentChannels: [], receivedTypes: [], rpcRefusals: [], acknowledgedWorkspaceIds: [], errors: [], closed: false }; log.webSockets.push(socket)
    ws.on('framesent', ({ payload }) => { try { const data = JSON.parse(String(payload)); if (typeof data.channel === 'string') { socket.sentChannels.push(data.channel); if (data.id) requests.set(data.id, data.channel) } } catch {} })
    ws.on('framereceived', ({ payload }) => { try { const data = JSON.parse(String(payload)); if (typeof data.type === 'string') socket.receivedTypes.push(data.type); if (data.type === 'handshake_ack') socket.acknowledgedWorkspaceIds.push(typeof data.workspaceId === 'string' ? data.workspaceId : null); if (data.error) socket.rpcRefusals.push({ channel: data.channel || requests.get(data.id), code: data.error.code, message: redact(data.error.message || '') }) } catch {} })
    ws.on('socketerror', error => socket.errors.push(redact(error))); ws.on('close', () => { socket.closed = true })
  })
  return log
}

    const text = translated(pKeys[route]); const visible = await page.locator('p').filter({ hasText: text }).first().isVisible()
    return { rendered: visible, text, kind: 'main detail picker paragraph' }
  }
  const buttonKeys = { projects: 'projectsList.addProject', pages: 'pages.newPage' }
  if (buttonKeys[route]) {
    const text = translated(buttonKeys[route]); const visible = await page.getByRole('button', { name: text, exact: true }).first().isVisible()
    return { rendered: visible, text, kind: 'surface-specific action button; no action submitted' }
  }
  const headerKey = route.startsWith('settings/') ? 'settings.' + (route.slice(9) === 'organizations' ? 'orgs' : route.slice(9)) + '.title'
    : ['dossier','radar','decisions','agents','focus'].includes(route) ? 'extraScreens.' + route + '.title'
    : route === 'search' ? 'searchPage.title' : null
  if (headerKey) {
    const text = translated(headerKey); const visible = await page.getByRole('heading', { name: text, exact: true }).first().isVisible()

async function finalCleanup() {
  const audit = { browserClosed: !browser, childAudits: [], profilesDeleted: [], errors: [] }
  if (browser) try { await within(browser.close(), 15_000, 'Owned browser cleanup timed out'); browser = null; audit.browserClosed = true } catch (error) { audit.errors.push(String(error)) }
  for (const child of children) {
    const childAudit = { pid: child.pid, initiallyExited: child.exitCode !== null || child.signalCode !== null, usedSigkill: false }
    const closed = childClosePromises.get(child)
    if (!childAudit.initiallyExited) {
      try { child.kill('SIGTERM') } catch {}
      try { await within(closed, 3_000, 'Owned child cleanup SIGTERM timed out') }
      catch { childAudit.usedSigkill = true; try { process.kill(-child.pid, 'SIGKILL') } catch { try { child.kill('SIGKILL') } catch {} } }
    }
    try { await within(closed, 5_000, 'Owned child did not close after cleanup'); childAudit.confirmedStopped = true; children.delete(child) }
    catch (error) { childAudit.confirmedStopped = false; audit.errors.push(String(error)) }
    childAudit.exitCode = child.exitCode; childAudit.signalCode = child.signalCode; audit.childAudits.push(childAudit)
  }
  for (const profile of profiles) try { await fsp.rm(profile, { recursive: true, force: true }); profiles.delete(profile); audit.profilesDeleted.push(profile) } catch (error) { audit.errors.push(String(error)) }
  try { await fsp.rm(path.join(runRoot, 'tmp'), { recursive: true, force: true }) } catch (error) { audit.errors.push(String(error)) }
  if (browserTmp) try { await fsp.rm(browserTmp, { recursive: true, force: true }); audit.shortBrowserTmpDeleted = true } catch (error) { audit.errors.push(String(error)) }
  audit.remainingChildren = children.size; audit.remainingProfiles = profiles.size; result.finalCleanup = audit
  if (audit.errors.length || children.size || profiles.size || !audit.browserClosed) result.errors.push({ phase: 'final cleanup', evidence: audit })
}
main().catch(error => { result.errors.push({ phase: 'runner', error: redact(error.stack || error) }); console.error(redact(error.stack || error)) }).finally(async () => {
  await finalCleanup()
  result.finishedAt = new Date().toISOString(); result.assertionsPassed = result.assertions.filter(a => a.pass).length; result.assertionsFailed = result.assertions.filter(a => !a.pass).length
  checkpoint(); console.log(`UI OBSERVATION DONE: ${result.assertionsPassed} passed checks, ${result.assertionsFailed} failed checks, ${result.errors.length} phase errors`)
  process.exitCode = result.errors.length || result.assertionsFailed ? 1 : 0
})
