import { readFileSync, writeFileSync } from 'node:fs'
import { createHash } from 'node:crypto'

const fixtureRoot = '/tmp/rox-core-integration-gate-ultra-20260930'
const testRuntime = process.env.ROX_CORE_GATE_BUN ?? `${fixtureRoot}/toolchain/package/bin/bun`
const sha256 = (bytes: Uint8Array | string) => createHash('sha256').update(bytes).digest('hex')
const records: object[] = []

function run(fixture: string, name: string, paths: string[], testName?: string) {
  const argv = [testRuntime, 'test', ...paths, ...(testName ? ['--test-name-pattern', testName] : [])]
  const result = Bun.spawnSync(argv, { cwd: `${fixtureRoot}/${fixture}`, stdout: 'pipe', stderr: 'pipe' })
  const log = Buffer.concat([result.stdout, result.stderr])
  const logPath = `${fixtureRoot}/evidence/${name}.log`
  writeFileSync(logPath, log)
  const record = { fixture, name, argv, exitCode: result.exitCode, logPath, logSha256: sha256(log) }
  records.push(record)
  return record
}

function mutant(fixture: string, name: string, path: string, before: string, after: string,
  expectedCount: number, testPaths: string[], testName: string, expectedExit: 'fail' | 'pass') {
  const filePath = `${fixtureRoot}/${fixture}/${path}`
  const original = readFileSync(filePath)
  const text = original.toString()
  if (text.split(before).length - 1 !== expectedCount) throw new Error(`${name}: unexpected source match count`)
  const changed = text.replaceAll(before, after)
  let record: ReturnType<typeof run>
  try {
    writeFileSync(filePath, changed)
    record = run(fixture, name, testPaths, testName)
  } finally {
    writeFileSync(filePath, original)
    if (sha256(readFileSync(filePath)) !== sha256(original)) throw new Error(`${name}: restore mismatch`)
  }
  if ((record.exitCode === 0) !== (expectedExit === 'pass')) throw new Error(`${name}: unexpected oracle outcome`)
  records.push({ name, mutationPath: path, originalSha256: sha256(original), mutationSha256: sha256(changed), restoredSha256: sha256(readFileSync(filePath)), expectedExit })
}
