function tree(root: string, relative = ''): { path: string; sha256: string; bytes: number }[] {
  const result: { path: string; sha256: string; bytes: number }[] = []
  for (const entry of readdirSync(`${root}/${relative}`, { withFileTypes: true })) {
    if (entry.name === 'node_modules' || entry.name === '.git') continue
    const path = relative ? `${relative}/${entry.name}` : entry.name
    if (entry.isSymbolicLink()) throw new Error(`unexpected source symlink ${path}`)
    if (entry.isDirectory()) result.push(...tree(root, path))
    else if (entry.isFile()) {
      const bytes = readFileSync(`${root}/${path}`)
      result.push({ path, sha256: sha256(bytes), bytes: bytes.length })
    }
  }
  return result.sort((a, b) => a.path.localeCompare(b.path, 'en'))
}
